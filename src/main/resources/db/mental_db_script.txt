CREATE TABLE user (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,

    username VARCHAR(255) NOT NULL UNIQUE,

    password_hash VARCHAR(255) NOT NULL,

    first_name VARCHAR(255) NOT NULL,

    last_name VARCHAR(255),

    gender VARCHAR(50),

    age_bucket VARCHAR(50),
    role VARCHAR(50),
    created_at TIMESTAMP(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6)
);


-- ============================================================
-- USER SETTINGS TABLE
-- ============================================================

CREATE TABLE user_settings (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,

    user_id BIGINT NOT NULL UNIQUE,

    email_notification BOOLEAN DEFAULT TRUE,

    reminder_enabled BOOLEAN DEFAULT TRUE,

    reminder_frequency VARCHAR(50),

    theme VARCHAR(50),

    ai_insights_enabled BOOLEAN DEFAULT TRUE,

    created_at TIMESTAMP(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),

    updated_at TIMESTAMP(6),

    CONSTRAINT fk_user_settings_user
        FOREIGN KEY (user_id)
        REFERENCES user(id)
        ON DELETE CASCADE
);


-- ============================================================
-- MOOD CHECK-IN TABLE
-- ============================================================
DROP TABLE mood_checkin;
CREATE TABLE mood_checkin (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,

    user_id BIGINT NOT NULL,

    mood_score INT NOT NULL,

    notes VARCHAR(1000),

    checkin_date DATE NOT NULL,

    created_at TIMESTAMP(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),


    CONSTRAINT uk_mood_user_date
        UNIQUE(user_id, checkin_date),


    CONSTRAINT fk_mood_user
        FOREIGN KEY (user_id)
        REFERENCES user(id)
        ON DELETE CASCADE,


    CONSTRAINT chk_mood_score
        CHECK (mood_score BETWEEN 1 AND 10)
);

ALTER TABLE mentaldb.mood_checkin
ADD COLUMN sleep_hours INT DEFAULT 0,
ADD COLUMN energy_level INT DEFAULT 0;

-- ============================================================
-- ASSESSMENT RESULT TABLE
-- ============================================================

CREATE TABLE assessment_result (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,


    user_id BIGINT NOT NULL,


    assessment_type VARCHAR(50) NOT NULL,


    score INT NOT NULL,


    severity_level VARCHAR(100),


    recommendation VARCHAR(2000),


    completed_at TIMESTAMP(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),


    CONSTRAINT fk_assessment_user
        FOREIGN KEY (user_id)
        REFERENCES user(id)
        ON DELETE CASCADE
);


-- ============================================================
-- JOURNAL ENTRY TABLE
-- ============================================================

CREATE TABLE journal_entry (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,


    user_id BIGINT NOT NULL,


    title VARCHAR(255) NOT NULL,


    content TEXT NOT NULL,


    sentiment VARCHAR(50),


    emotion VARCHAR(50),


    entry_date DATE,


    created_at TIMESTAMP(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),


    CONSTRAINT fk_journal_user
        FOREIGN KEY (user_id)
        REFERENCES user(id)
        ON DELETE CASCADE
);



-- ============================================================
-- EMERGENCY CONTACT TABLE
-- ============================================================

CREATE TABLE emergency_contact (

    id BIGINT AUTO_INCREMENT PRIMARY KEY,


    user_id BIGINT NOT NULL,


    contact_name VARCHAR(255) NOT NULL,


    relationship VARCHAR(100),


    phone_number VARCHAR(50) NOT NULL,


    email VARCHAR(255),


    primary_contact BOOLEAN DEFAULT FALSE,


    created_at TIMESTAMP(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),


    CONSTRAINT fk_emergency_user
        FOREIGN KEY (user_id)
        REFERENCES user(id)
        ON DELETE CASCADE
);



-- ============================================================
-- INDEXES FOR PERFORMANCE
-- ============================================================

CREATE INDEX idx_mood_user
ON mood_checkin(user_id);


CREATE INDEX idx_assessment_user
ON assessment_result(user_id);


CREATE INDEX idx_journal_user
ON journal_entry(user_id);


CREATE INDEX idx_emergency_user
ON emergency_contact(user_id);
